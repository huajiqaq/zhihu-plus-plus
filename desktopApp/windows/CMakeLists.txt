cmake_minimum_required(VERSION 3.24)
project(ZhihuWindowsNotifications LANGUAGES CXX)

if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The optional MSIX host requires MSVC and Windows x64.")
endif()
if(NOT DEFINED ZHIHU_PACKAGE_INPUTS OR NOT EXISTS "${ZHIHU_PACKAGE_INPUTS}/package_config.h")
    message(FATAL_ERROR "Run package-msix.ps1 with a project jpackage app-image first.")
endif()
find_package(JNI REQUIRED)
add_executable(ZhihuNotifications WIN32 WindowsNotifications.cpp)
target_compile_features(ZhihuNotifications PRIVATE cxx_std_20)
target_include_directories(ZhihuNotifications PRIVATE ${JNI_INCLUDE_DIRS} "${ZHIHU_PACKAGE_INPUTS}")
target_compile_definitions(ZhihuNotifications PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(ZhihuNotifications PRIVATE /W4 /WX /permissive- /EHsc /utf-8 /guard:cf)
target_link_options(ZhihuNotifications PRIVATE /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA /guard:cf)
set_property(TARGET ZhihuNotifications PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_link_libraries(ZhihuNotifications PRIVATE windowsapp ole32 advapi32 bcrypt shell32)
