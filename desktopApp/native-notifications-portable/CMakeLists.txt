cmake_minimum_required(VERSION 3.24)
project(ZhihuPortableNotifications LANGUAGES CXX)
find_package(JNI REQUIRED)
if(WIN32 AND MSVC)
    add_library(ZhihuPortableNotifications SHARED WindowsPortableNotifications.cpp)
    add_executable(ZhihuNotificationActivator WIN32 WindowsPortableNotifications.cpp)
    target_compile_definitions(ZhihuNotificationActivator PRIVATE ZHIHU_NOTIFICATION_ACTIVATOR)
    foreach(target ZhihuPortableNotifications ZhihuNotificationActivator)
        target_compile_features(${target} PRIVATE cxx_std_20)
        target_include_directories(${target} PRIVATE ${JNI_INCLUDE_DIRS})
        target_compile_definitions(${target} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
        target_compile_options(${target} PRIVATE /W4 /permissive- /EHsc /utf-8 /guard:cf)
        target_link_options(${target} PRIVATE /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA /guard:cf)
        set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
        target_link_libraries(${target} PRIVATE windowsapp ole32 advapi32 shell32)
    endforeach()
elseif(APPLE)
    enable_language(OBJCXX)
    add_library(ZhihuPortableNotifications SHARED MacosPortableNotifications.mm)
    target_compile_features(ZhihuPortableNotifications PRIVATE cxx_std_20)
    target_include_directories(ZhihuPortableNotifications PRIVATE ${JNI_INCLUDE_DIRS})
    target_compile_options(ZhihuPortableNotifications PRIVATE -fobjc-arc -Wall -Wextra)
    target_link_libraries(ZhihuPortableNotifications PRIVATE "-framework Foundation" "-framework UserNotifications" "-framework AppKit")
else()
    message(FATAL_ERROR "Portable notification bridge targets Windows MSVC and macOS only; Linux uses D-Bus.")
endif()
